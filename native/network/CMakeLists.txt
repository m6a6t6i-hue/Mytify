cmake_minimum_required(VERSION 3.22)
project(mytify_network LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(FETCHCONTENT_SOURCE_DIR_MBEDTLS "${CMAKE_CURRENT_SOURCE_DIR}/mbedtls" CACHE PATH "" FORCE)
set(AIRPLAY_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
option(MYTIFY_NETWORK_TESTS "Run protocol tests" OFF)
set(AIRPLAY_BUILD_TESTS ${MYTIFY_NETWORK_TESTS} CACHE BOOL "" FORCE)
set(LIBSAMPLERATE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(airplay)
add_subdirectory(libsamplerate)
find_package(Threads REQUIRED)
add_library(mytify_airplay SHARED bridge.cpp)
target_link_libraries(mytify_airplay PRIVATE raop_sender samplerate Threads::Threads)
target_compile_definitions(mytify_airplay PRIVATE MYTIFY_EXPORTS)
if(ANDROID)
 target_sources(mytify_airplay PRIVATE jni.cpp)
endif()
if(MYTIFY_NETWORK_TESTS)
 enable_testing()
 add_test(NAME airplay_protocol COMMAND raop_core_tests)
 add_test(NAME airplay_loopback COMMAND raop_loop_tests)
endif()
